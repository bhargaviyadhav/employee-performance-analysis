1. Find employees with salary above 50000.

SELECT Name, Salary
FROM Employees
WHERE Salary > 50000;


2. Display employees in ascending order of salary.

SELECT Name, Salary
FROM Employees
ORDER BY Salary ASC;


3. Find employees working in Bangalore.

SELECT Name, City
FROM Employees
WHERE City = 'Bangalore';


4. Count employees in each department.

SELECT Department, count()
FROM Employees
GROUP BY Department;


5. Find the average salary of employees.

SELECT Name, avg(Salary)
FROM Employees;


6. Find employees with experience above 2 years.

SELECT  Name ,Experience
FROM Employees
WHERE Experience > 2 ;
